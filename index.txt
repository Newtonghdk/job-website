  <script>
    /*
      IMPORTANT:
      Replace the value below with your Google Apps Script Web App URL.

      Example:
      const API_URL =
      "https://script.google.com/macros/s/ABC123/exec";
    */

    const API_URL = "https://script.google.com/macros/s/AKfycbw75taqFAJhumaE9wDWD00meETuvVL-Eu_qV8LPE9gHDS-QgwIFjEbG8TKSBP_hmb-_qQ/exec";

    /*
      These sample jobs appear until you connect Google Sheets.
    */

    const sampleJobs = [
      {
        ID: "job-001",
        Title: "District Program Manager",
        Company: "Tata Steel Foundation",
        Location: "Jharkhand, Odisha",
        Skills: "3–5 years",
        "Date Posted": "18-07-2026",
        URL: "job.html?id=job-001"
      },
      {
        ID: "job-002",
        Title: "Education Program Officer",
        Company: "ABC Foundation",
        Location: "Remote",
        Skills: "2–4 years",
        "Date Posted": "25-07-2026",
        URL: "job.html?id=job-002"
      }
    ];

    let allJobs = [];

    function escapeHTML(value) {
      return String(value || "").replace(/[&<>"']/g, function(character) {
        return {
          "&": "&amp;",
          "<": "&lt;",
          ">": "&gt;",
          '"': "&quot;",
          "'": "&#039;"
        }[character];
      });
    }

    function renderJobs(jobs) {
      const jobList = document.getElementById("job-list");

      if (!jobs.length) {
        jobList.innerHTML = "<p>No jobs found.</p>";
        return;
      }

      jobList.innerHTML = jobs.map(function(job) {
        const jobURL = job.URL || ("job.html?id=" + job.ID);

        return `
          <a class="job-card" href="${escapeHTML(jobURL)}">

            <div class="card-top">
              <div>
                <div class="organization">
                  ${escapeHTML(job.Company)}
                </div>

                <h3>
                  ${escapeHTML(job.Title)}
                </h3>
              </div>

              <div class="logo-box">
                Logo
              </div>
            </div>

            <div class="job-meta">
              <div>
                <span>⌖</span>
                ${escapeHTML(job.Location)}
              </div>

              <div>
                <span>▣</span>
                ${escapeHTML(job.Skills || "Experience not specified")}
              </div>

              <div>
                <span>◷</span>
                ${escapeHTML(job["Date Posted"] || "")}
              </div>
            </div>

            <div class="view-job">
              View Job →
            </div>

          </a>
        `;
      }).join("");
    }

    function filterJobs() {
      const searchTerm =
        document.getElementById("search-term").value.toLowerCase();

      const searchLocation =
        document.getElementById("search-location").value.toLowerCase();

      const category =
        document.getElementById("category-filter").value.toLowerCase();

      const location =
        document.getElementById("location-filter").value.toLowerCase();

      const experience =
        document.getElementById("experience-filter").value.toLowerCase();

      const employmentType =
        document.getElementById("type-filter").value.toLowerCase();

      const filteredJobs = allJobs.filter(function(job) {
        const jobText = Object.values(job).join(" ").toLowerCase();

        return (
          jobText.includes(searchTerm) &&
          jobText.includes(searchLocation) &&
          jobText.includes(category) &&
          jobText.includes(location) &&
          jobText.includes(experience) &&
          jobText.includes(employmentType)
        );
      });

      renderJobs(filteredJobs);
    }

    document
      .getElementById("search-form")
      .addEventListener("submit", function(event) {
        event.preventDefault();
        filterJobs();
      });

    document
      .getElementById("category-filter")
      .addEventListener("change", filterJobs);

    document
      .getElementById("location-filter")
      .addEventListener("change", filterJobs);

    document
      .getElementById("experience-filter")
      .addEventListener("change", filterJobs);

    document
      .getElementById("type-filter")
      .addEventListener("change", filterJobs);

    document
      .getElementById("clear-filters")
      .addEventListener("click", function() {
        document.getElementById("search-term").value = "";
        document.getElementById("search-location").value = "";
        document.getElementById("category-filter").value = "";
        document.getElementById("location-filter").value = "";
        document.getElementById("experience-filter").value = "";
        document.getElementById("type-filter").value = "";

        renderJobs(allJobs);
      });

    /*
      Load jobs from Google Sheets through Apps Script.
      If the URL is not added or fails, sample jobs are shown.
    */

    if (API_URL.indexOf("http") === 0) {
      fetch(API_URL)
        .then(function(response) {
          return response.json();
        })
        .then(function(data) {
          allJobs = data;
          renderJobs(allJobs);
        })
        .catch(function() {
          allJobs = sampleJobs;
          renderJobs(allJobs);
        });
    } else {
      allJobs = sampleJobs;
      renderJobs(allJobs);
    }
  </script>
